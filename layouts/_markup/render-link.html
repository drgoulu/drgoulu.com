{{/* Hugo Markdown render hook for links with Wikipedia shortcut and auto-styling support */}} {{- $dest := .Destination -}} {{- $text := .Text -}} {{- $title := .Title -}} {{- $is_wiki := false -}} {{- if findRE `^/\d{4}/\d{4}-\d{2}-\d{2}-` $dest -}} {{- $dest = replaceRE `^/(\d{4})/\d{4}-(\d{2})-(\d{2})-(.*?)/?$` `/$1/$2/$3/$4/` $dest -}} {{- end -}} {{- if or (strings.HasPrefix $dest "w:") (strings.HasPrefix $dest "wiki:") -}} {{- $is_wiki = true -}} {{- $raw := strings.TrimPrefix "wiki:" (strings.TrimPrefix "w:" $dest) -}} {{- $raw = replace $raw "\\" "" -}} {{- $lang := "fr" -}} {{- $article := "" -}} {{- if strings.Contains $raw ":" -}} {{- $parts := split $raw ":" -}} {{- $first := index $parts 0 -}} {{- if findRE `^[a-z]{2,3}$` $first -}} {{- $lang = $first -}} {{- $article = delimit (after 1 $parts) ":" -}} {{- else -}} {{- $article = $raw -}} {{- end -}} {{- else if strings.Contains $raw "/" -}} {{- $parts := split $raw "/" -}} {{- $first := index $parts 0 -}} {{- if findRE `^[a-z]{2,3}$` $first -}} {{- $lang = $first -}} {{- $article = delimit (after 1 $parts) "/" -}} {{- else -}} {{- $article = $raw -}} {{- end -}} {{- else if or (eq $raw "en") (eq $raw "de") (eq $raw "es") (eq $raw "it") -}} {{- $lang = $raw -}} {{- $article = "" -}} {{- else -}} {{- $article = $raw -}} {{- end -}} {{- if eq $article "" -}} {{- $article = .PlainText | default $text -}} {{- end -}} {{- $article = replace $article " " "_" -}} {{- $dest = printf "https://%s.wikipedia.org/wiki/%s" $lang $article -}} {{- else if in $dest "wikipedia.org" -}} {{- $is_wiki = true -}} {{- else if and (not (strings.HasPrefix $dest "http")) (not (strings.HasPrefix $dest "/")) (not (strings.HasPrefix $dest "#")) (not (strings.HasPrefix $dest "mailto:")) -}} {{- $clean_dest := strings.TrimPrefix "./" $dest -}} {{- $post_year := .Page.Date.Format "2006" -}} {{- $section := .Page.Section | default "posts" -}} {{- with resources.Get (path.Join $section $post_year $clean_dest) -}} {{- $dest = .RelPermalink -}} {{- else -}} {{- if fileExists (path.Join "content" $section $post_year $clean_dest) -}} {{- $dest = printf "/%s/%s/%s" $section $post_year $clean_dest -}} {{- end -}} {{- end -}} {{- end -}} {{ $text | safeHTML }}
